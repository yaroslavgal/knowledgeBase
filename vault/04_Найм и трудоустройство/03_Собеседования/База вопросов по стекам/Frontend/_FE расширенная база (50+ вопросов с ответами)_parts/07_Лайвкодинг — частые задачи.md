## Лайвкодинг — частые задачи

### 53. Debounce
```js
function debounce(fn, ms) {
  let timer;
  return function(...args) {
    clearTimeout(timer);
    timer = setTimeout(() => fn.apply(this, args), ms);
  };
}
```

### 54. Throttle
```js
function throttle(fn, ms) {
  let isThrottled = false;
  return function(...args) {
    if (isThrottled) return;
    fn.apply(this, args);
    isThrottled = true;
    setTimeout(() => { isThrottled = false; }, ms);
  };
}
```

### 55. Реализуй Promise.all `S`
```js
function promiseAll(promises) {
  return new Promise((resolve, reject) => {
    if (!promises.length) return resolve([]);
    const results = []; let count = 0;
    promises.forEach((p, i) => {
      Promise.resolve(p).then(
        v => { results[i] = v; if (++count === promises.length) resolve(results); },
        reject
      );
    });
  });
}
```
Ключевое: `Promise.resolve(p)` для не-промисов, `results[i]` сохраняет порядок, один reject отклоняет весь.

### 56. LRU Cache `S`
```js
class LRUCache {
  constructor(cap) { this.cap = cap; this.cache = new Map(); }
  get(k) {
    if (!this.cache.has(k)) return -1;
    const v = this.cache.get(k);
    this.cache.delete(k); this.cache.set(k, v);
    return v;
  }
  put(k, v) {
    if (this.cache.has(k)) this.cache.delete(k);
    this.cache.set(k, v);
    if (this.cache.size > this.cap) this.cache.delete(this.cache.keys().next().value);
  }
}
```
Трюк: Map сохраняет порядок вставки. Delete+set = перемещение в конец. `keys().next().value` = самый старый.

### 57. EventEmitter
```js
class EventEmitter {
  constructor() { this.l = new Map(); }
  on(e, fn) { (this.l.get(e) || this.l.set(e, []).get(e)).push(fn); return this; }
  off(e, fn) { const a = this.l.get(e); if (a) this.l.set(e, a.filter(f => f !== fn)); return this; }
  emit(e, ...a) { (this.l.get(e) || []).forEach(fn => fn(...a)); return this; }
  once(e, fn) { const w = (...a) => { fn(...a); this.off(e, w); }; return this.on(e, w); }
}
```

### 58. Реализуй bind
```js
Function.prototype.myBind = function(ctx, ...args) {
  const fn = this;
  return function(...newArgs) { return fn.apply(ctx, [...args, ...newArgs]); };
};
```

### 59. Deep clone
```js
function deepClone(o) {
  if (o === null || typeof o !== "object") return o;
  if (o instanceof Date) return new Date(o);
  if (Array.isArray(o)) return o.map(deepClone);
  const c = {};
  for (const k of Object.keys(o)) c[k] = deepClone(o[k]);
  return c;
}
// Или встроенный structuredClone(o) (2022+)
```

### 60. useDebounce hook
```jsx
function useDebounce(value, delay) {
  const [d, setD] = useState(value);
  useEffect(() => {
    const t = setTimeout(() => setD(value), delay);
    return () => clearTimeout(t);
  }, [value, delay]);
  return d;
}
```
